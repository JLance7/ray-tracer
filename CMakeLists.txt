cmake_minimum_required(VERSION 3.20)
project(IronVeil LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "This version uses the Windows desktop API.")
endif()
add_executable(IronVeil WIN32 src/main.c)
target_compile_features(IronVeil PRIVATE c_std_11)
target_compile_definitions(IronVeil PRIVATE UNICODE _UNICODE)
target_link_libraries(IronVeil PRIVATE user32 gdi32)
if(MINGW)
    target_link_options(IronVeil PRIVATE -municode -static -static-libgcc)
    target_link_libraries(IronVeil PRIVATE m)
endif()
